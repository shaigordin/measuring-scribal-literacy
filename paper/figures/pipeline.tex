% ---------------- research process
\foreach \i/\t [count=\n from 0] in {

% ---------------- pipeline
\def\Y{-8mm}
\node[stage=data] (s0) at (0*\W, \Y) {\textbf{\color{data}1 Corpus}\\[2pt]
  TLHdig 0.3, normalised words and signs per manuscript.\\[2pt]
  \Q{corpus.reference_texts} reference manuscripts; \Q{corpus.orphans_total} orphans (CTH~470, EBo).};
\node[stage=data] (s1) at (1*\W, \Y) {\textbf{\color{data}2 Reference}\\[2pt]
  \Q{trad.count} traditions, each defined by a published study and mapped to Laroche's CTH.\\[2pt]
  Unclaimed CTH ranges kept as residue.};
\node[stage=data] (s2) at (2*\W, \Y) {\textbf{\color{data}3 Features}\\[2pt]
  tf-idf of word forms (\emph{what} is written) and of sign n-grams within words (\emph{how}).\\[2pt]
  Each word tagged with a word class.};
\node[stage=model] (s3) at (3*\W, \Y) {\textbf{\color{model}4 Score}\\[2pt]
  Cosine to each tradition's centroid.\\[2pt]
  margin $=s_1-s_2$, the lead of the nearest over the runner-up.};
\node[stage=model] (s4) at (4*\W, \Y) {\textbf{\color{model}5 Decide}\\[2pt]
  Assign if margin $\geq \Q{E2.margin}$ (\Q{E2.precision} precision on held-out windows), else abstain.\\[2pt]
  Outside genres as classes.};
\node[stage=out] (s5) at (5*\W, \Y) {\textbf{\color{out}6 Output}\\[2pt]
  \Q{E5.assigned} of \Q{corpus.orphans_used} orphans placed.\\[2pt]
  Each placement with its margin split by word class and layer, for checking on the tablet.};
\foreach \a/\b in {s0/s1, s1/s2, s2/s3, s3/s4, s4/s5} \draw[flow] (\a.east) -- (\b.west);
\node[lane] at (-2mm, \Y-12mm) {Pipeline};
